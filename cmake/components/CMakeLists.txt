cmake_minimum_required(VERSION 3.24)
project(LLVMStaticComponents LANGUAGES C CXX)
find_package(LLVM REQUIRED CONFIG)
if(NOT LLVM_PACKAGE_VERSION VERSION_EQUAL SDK_LLVM_VERSION)
  message(FATAL_ERROR "Unexpected LLVM version: ${LLVM_PACKAGE_VERSION}")
endif()
llvm_map_components_to_libnames(pending ${SDK_COMPONENTS})
set(selected)
while(pending)
  list(POP_FRONT pending component)
  if(component IN_LIST selected)
    continue()
  endif()
  if(NOT TARGET ${component})
    message(FATAL_ERROR "Missing LLVM component: ${component}")
  endif()
  get_target_property(kind ${component} TYPE)
  if(NOT kind STREQUAL "STATIC_LIBRARY")
    message(FATAL_ERROR "Expected a static library: ${component} (${kind})")
  endif()
  list(APPEND selected ${component})
  get_target_property(dependencies ${component} INTERFACE_LINK_LIBRARIES)
  if(dependencies)
    # LLVM's exported component edges can be wrapped in LINK_ONLY expressions.
    string(REGEX MATCHALL "LLVM[A-Za-z0-9_]+" llvm_dependencies "${dependencies}")
    list(APPEND pending ${llvm_dependencies})
  endif()
endwhile()
list(SORT selected)
file(WRITE "${CMAKE_BINARY_DIR}/components.txt" "${selected};llvm-headers;cmake-exports")

cmake_minimum_required(VERSION 3.24)
project(LLVMStaticSmoke LANGUAGES C CXX)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL SDK_POINTER_SIZE)
  message(FATAL_ERROR "Smoke consumer architecture does not match SDK")
endif()
find_package(LLVM REQUIRED CONFIG)
if(NOT LLVM_PACKAGE_VERSION VERSION_EQUAL SDK_LLVM_VERSION)
  message(FATAL_ERROR "Unexpected LLVM SDK version")
endif()
llvm_map_components_to_libnames(llvm_libraries core orcjit passes nativecodegen)
foreach(library IN LISTS llvm_libraries)
  get_target_property(kind ${library} TYPE)
  if(NOT kind STREQUAL "STATIC_LIBRARY")
    message(FATAL_ERROR "Shared LLVM component: ${library}")
  endif()
endforeach()
add_library(sdk_smoke SHARED jit.cpp)
target_compile_features(sdk_smoke PRIVATE cxx_std_17)
target_include_directories(sdk_smoke SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
target_compile_definitions(sdk_smoke PRIVATE LLVM_BUILD_STATIC)
target_link_libraries(sdk_smoke PRIVATE ${llvm_libraries})
set_target_properties(sdk_smoke PROPERTIES CXX_VISIBILITY_PRESET hidden)
add_executable(sdk_smoke_test main.cpp)
target_link_libraries(sdk_smoke_test PRIVATE sdk_smoke)
if(APPLE)
  set_target_properties(sdk_smoke PROPERTIES
    INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR ON)
  set_target_properties(sdk_smoke_test PROPERTIES
    INSTALL_RPATH "@loader_path" BUILD_WITH_INSTALL_RPATH ON)
endif()
enable_testing()
add_test(NAME jit_from_shared_library COMMAND sdk_smoke_test)
